ИГРА КРАШ --- let crashState = 'idle'; // idle, flying, crashed let crashMult = 1.00; let crashBetAmount = 0; let crashAutoCashout = 0; let crashInterval = null; let canvas, ctx; let rocketX = 0, rocketY = 0; // Инициализация Canvas function initCrashCanvas() { canvas = document.getElementById('crashCanvas'); if (!canvas) return; ctx = canvas.getContext('2d'); canvas.width = canvas.offsetWidth; canvas.height = canvas.offsetHeight; // Рисуем пустой график drawGraph(); } function drawGraph() { if (!ctx) return; ctx.clearRect(0, 0, canvas.width, canvas.height); ctx.strokeStyle = 'rgba(52, 211, 153, 0.2)'; ctx.lineWidth = 1; for (let i = 0; i < canvas.width; i += 40) { ctx.beginPath(); ctx.moveTo(i, 0); ctx.lineTo(i, canvas.height); ctx.stroke(); } for (let i = 0; i < canvas.height; i += 40) { ctx.beginPath(); ctx.moveTo(0, i); ctx.lineTo(canvas.width, i); inp.value = Math.floor(parseFloat(inp.value) * 2); } // Главная кнопка (Ставка / Забрать) function crashAction() { if (crashState === 'idle') { startCrashBet(); } else if (crashState === 'flying') { cashoutCrash(); } } function startCrashBet() { const bet = parseFloat(document.getElementById('crashBet').value); if (!bet || bet < 10) return alert('Минимальная ставка — 10 ⭐'); if (bet > balance) return alert('Недостаточно средств!'); document.getElementById('crashBtn').classList.add('win'); document.getElementById('crashStatus').textContent = 'Ракета в полёте...'; // Генерация случайного хэша (для честности) document.getElementById('crashHash').textContent = '#' + Math.random().toString(16).slice(2, 10); // Логика падения (Твоя подкрутка) const isBankSmall = balance < 500; const isBankBig = balance > 1500; let crashPoint = 1.00; const rand = Math.random(); if (isBankSmall) { // Если банк мал, ракета летит далеко (до 52x) if (rand < 0.15) crashPoint = 1.00; // 15% шанс краша на 1.0 else if (rand < 0.4) crashPoint = 1.5 + Math.random() * 3; else crashPoint = 5 + Math.random() * 47; } else if (isBankBig) { // Если банк большой, ракета падает быстро (0x или 1.5x) if (rand < 0.5) crashPoint = 1.00; // 50% шанс краша на 1.0 (0x) else if (rand < 0.8) crashPoint = 1.10 + Math.random() * 0.4; // до 1.5x else crashPoint = 1.5 + Math.random() * 3; // редко больше 12x } else { // Средний банк (500-1500) if (rand < 0.3) crashPoint = 1.00; else if (rand < 0.6) crashPoint = 1.2 + Math.random() * 2; else crashPoint = 3 + Math.random() * 9; } // Запускаем анимацию let startTime = Date.now(); const flightDuration = 3000 + Math.random() * 2000; // 3-5 секунд полёта crashInterval = setInterval(() => { const elapsed = Date.now() - startTime; const progress = Math.min(elapsed / flightDuration, 1); // Множитель растёт crashMult = 1.00 + (crashPoint - 1.00) * Math.pow(progress, 1.5); // Если долетели до точки краша if (crashMult >= crashPoint) { crashMult = crashPoint; endCrash(false); return; } // Обновляем интерфейс document.getElementById('crashMult').textContent = crashMult.toFixed(2) + 'x'; const maxY = 20; rocketX = 50 + (maxX - 50) * progress; rocketY = 150 - (150 - maxY) * progress; // Тряска ракеты const shake = Math.sin(elapsed / 50) * 2; const rocketEl = document.getElementById('rocket'); rocketEl.style.left = (rocketX + shake) + 'px'; rocketEl.style.bottom = (canvas.height - rocketY) + 'px'; // Рисуем график (линию) drawCrashLine(progress); }, 50); } function drawCrashLine(progress) { if (!ctx) return; p += 0.01) { const x = 50 + (maxX - 50) * p; const y = canvas.height - (150 - (150 - maxY) * p); if (p === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y); } ctx.stroke(); ctx.shadowBlur = 0; } function cashoutCrash() { if (crashState !== 'flying') return; document.getElementById('rocket').style.bottom = '20px'; document.getElementById('crashMult').textContent = '1.00x'; if (ctx) drawGraph(); }, 1500); } function endCrash(isWin) { clearInterval(crashInterval);
